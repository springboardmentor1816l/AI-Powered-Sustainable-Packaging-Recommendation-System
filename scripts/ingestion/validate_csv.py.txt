import pandas as pd
import os

# Input paths
materials_path = "data/raw_datasets/materials/materials.csv"
products_path = "data/raw_datasets/products/products.csv"

# Output paths
materials_out = "data/processed/materials_clean.csv"
products_out = "data/processed/products_clean.csv"

# Create output folder if missing
os.makedirs("data/processed", exist_ok=True)

# -------- MATERIALS VALIDATION --------
materials = pd.read_csv(materials_path)

print("\n--- MATERIALS INFO ---")
print(materials.info())
print("\nMissing Values:\n", materials.isnull().sum())

# Remove duplicates
materials = materials.drop_duplicates()

# Save cleaned file
materials.to_csv(materials_out, index=False)
print("\n✅ Cleaned materials saved to:", materials_out)

# -------- PRODUCTS VALIDATION --------
products = pd.read_csv(products_path)

print("\n--- PRODUCTS INFO ---")
print(products.info())
print("\nMissing Values:\n", products.isnull().sum())

# Remove duplicates
products = products.drop_duplicates()

# Save cleaned file
products.to_csv(products_out, index=False)
print("\n✅ Cleaned products saved to:", products_out)
